// INDICADOR BULLEX - SINAL NO TEMPO CERTO var podeEntrar = false; var ultimoSinal = ""; function verificarTempoCerto() { var agora = new Date(); var segundos = agora.getSeconds(); // TEMPO CERTO DA BULLEX - 1 minuto // A entrada tem que ser entre 57s e 59s pra pegar a próxima vela if (segundos >= 57 && segundos <= 59) { if (podeEntrar == false) { // pra não repetir var r = detectarPuxadaCompra(window.candles); if (r.valido && r.pct >= 84) { ultimoSinal = r.dir; mostrarSinal(r); podeEntrar = true; // Toca som de alerta try { navigator.vibrate(500); } catch(e){} } } } else if (segundos >= 0 && segundos <= 2) { // Começou a nova vela - libera de novo podeEntrar = false; } // Atualiza o cronômetro na tela var falta = 60 - segundos; document.getElementById('cronometro').innerText = falta + 's'; } function mostrarSinal(resultado) { var el = document.getElementById('sinal'); el.innerText = resultado.dir + ' ' + resultado.pct + '% - ENTRE AGORA!'; el.className = resultado.dir == 'CALL' ? 'call piscando' : 'put piscando'; // Se for CALL, verde. PUT, vermelho. if (resultado.dir == 'CALL') { el.style.background = '#00ff88'; el.innerHTML = '▲ COMPRA
' + resultado.pct + '%
' + resultado.msg + ''; } else { el.style.background = '#ff3b3b'; el.innerHTML = '▼ VENDA
' + resultado.pct + '%
' + resultado.msg + ''; } } // RODA A CADA 500ms - verifica o tempo certo setInterval(verificarTempoCerto, 500);
